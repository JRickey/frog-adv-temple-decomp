# sub_08011E40 (+ sub_08011F3C) — deferred

The asm slice `asm/disasm_0x08011e40.s` [0x08011e40, 0x08012098) (600 bytes)
contains **two** functions:

- `sub_08011E40` (0x11e40-0x11f0e, 240 bytes): BL-called from 0x801240a /
  0x8012430 / 0x801245c. Returns `u8`.
- `sub_08011F3C` (0x11f3c-0x12082, ~360 bytes): NO static BL caller and not a
  pointer word anywhere in ROM — reached indirectly / dead. Signature
  `(const BlitRecord *records, u8 idx)`.

Both are fully understood structurally; the residual is **agbcc register
coloring / stack-spill** divergence, which is the permuter's job. The
permuter `.venv` is a broken self-referential symlink in **main** (not just the
worktree) so it could not run this pass. Resume here once the permuter is
repaired — both functions are NEAR-match and pure-C-matchable (NOT
STRONG_UNMATCHABLE; no class3-libgcc prologue).

## Drift

### sub_08011E40 — best byte_diff 199, diff_count 64 (clean) / 58 (pinned)
Structure MATCHES (DMA double-fill, BLDCNT/BLDALPHA writes, the bit-scan loop,
the tail `sub_08011AA4` call are all byte-aligned). Residual is pure coloring:
- baserom loads `gIwram_6110` base into **r0** (scratch) for the pre-loop
  `flags[0x2a] != flags[0x2b]` compare, THEN `adds r7,r0,#0` copies it to a
  callee-saved reg for the loop. My C loads straight into r7. agbcc's
  scratch-then-promote choice was not reproduced by any shape tried.
- baserom hoists the loop-invariant constants `1`→**r8** and `0`→**r9**; my C
  swaps them (`1`→r9, `0`→r8). Pinning `register s32 one asm("r8")=1;` +
  `register s32 zeroc asm("r9")=0;` (plus `result`→sl, `i`→r4, `dma`→r5) got it
  to diff_count 58 but the constant order + the DMA first-block store order
  (`src=sp; dst; cnt` vs my reorder) still diverge.
- `result` lives in **sl** the whole function; pinning `register u8 result
  asm("sl")` reproduces that.
- Loop is a `do { ... i=(u16)(i+1); } while (i<=1)` (bottom-tested, u16
  increment) — this WAS reproduced (top-tested `for` was worse).

Levers tried: OLD_AGBCC (default) and AGBCC per-TU swap (no change);
`-fno-gcse`, `-fno-schedule-insns[2]`, `-fno-strength-reduce`,
`-fno-expensive-optimizations` (no change); full register pins incl. high regs
(58, but introduces other coloring drift); a/b-local vs goto-continue
phrasings; reading the pre-compare bytes via direct `((u8*)&gIwram_6110)[..]`
to force scratch (worse — CSE merged the two address loads).
Permuter NOT run (venv broken). Next: run permuter from the pinned-58 base.

### sub_08011F3C — best byte_diff 371, diff_count 154
Structure MATCHES (the `idx*8 -idx *4` stride-28 record arithmetic, the
`sl=records` param hold, `next=rec+1`, the `_field_34 = next->_field_00` read
via `rec[0x1c]` addressing, the two tile-copy loop nests, the final
`rec->_field_06 * next->_field_06 + next->tilesPerCol` are all byte-aligned).
The **HYBRID recompute** shape was the key: cache `rec`/`next` within each of
the 3 phases but RE-derive `&records[idx]` at the start of each phase (setup,
loop-B-prep, final) — this reproduced `sl=records` + per-phase rec recompute.
Caching rec/next once across all phases (sl=rec) scored 197; recomputing every
field access scored 195; the hybrid scores 154.
Residual:
- `sub sp,#12` vs baserom `sub sp,#8` — one extra stack spill slot (baserom
  keeps only `idx*8`@sp[0] and the loop-A `oy`@sp[4]; mine spills an extra
  value). Reducing live locals / counter scope is the next lever.
- `idx` in r8 vs baserom r9; `rec`/`next` in r5/r6 vs baserom r2/r4 — coloring.
The first ~8 diff lines (0x0000-0x000e) are the prior function's `.word` pool
mis-shown as instructions; ignore them.

/* Graphics-blit descriptor block at 0x03006480. The same block is set up
 * by sub_08011984; only the fields touched here are modelled. */
struct BlitDesc {
    u8 state;     /* +0x00 */
    u8 _pad01;
    u8 _field_02; /* +0x02: written with the active sub-channel index */
    u8 _pad03[5];
    u8 mode;      /* +0x08 */
    u8 _field_09; /* +0x09 */
    u8 _pad0a;
    u8 _field_0b; /* +0x0b: copied from record byte +0x0c */
    u8 _pad0c[4];
    const u16 *src; /* +0x10: ROM tile source */
    u8 _pad14[4];
    u16 *dst; /* +0x18: EWRAM tile destination */
    u8 _pad1c[16];
    u16 _field_2c; /* +0x2c */
    u16 _field_2e; /* +0x2e */
    u16 width;     /* +0x30 */
    u16 height;    /* +0x32 */
    u16 _field_34; /* +0x34 */
    u16 _field_36; /* +0x36 */
};

/* Tile-grid geometry block at 0x03006540 (sub-grid width/height). */
struct GridDesc {
    u8 _pad00[48];
    u16 width;  /* +0x30 */
    u16 height; /* +0x32 */
};

/* Scroll/viewport state at 0x030060a0; +0x1a is the screen-row stride. */
struct ViewportDesc {
    u8 _pad00[26];
    u16 rowStride; /* +0x1a */
};

/* Per-scene blit record (stride 0x1c). sub_08011F3C consumes record[idx],
 * record[idx+1] and record[idx+2]. */
struct BlitRecord {
    u16 _field_00;
    u16 tilesPerCol; /* +0x02 */
    u16 _field_04;
    u16 _field_06;
    u8 _pad08[4];
    u8 _field_0c; /* +0x0c */
    u8 _pad0d[3];
    const u16 *src; /* +0x10 */
    u8 _pad14[8];   /* pad to the 0x1c record stride */
};

extern struct GridDesc gIwram_6540;
extern struct ViewportDesc gIwram_60A0;

u8 sub_08011E40(void)
{
    u8 *base;
    DmaChannel *dma;
    u16 zero;
    s32 i;
    u8 result;

    result = 0;
    base = (u8 *)&gIwram_6110;

    if (base[0x2a] != base[0x2b]) {
        i = 0;
        dma = &REG_DMA3;
        do {
            if ((base[0x2a] >> i & 1) == 0)
                goto next;
            if ((base[0x2a] >> i & 1) == (base[0x2b] >> i & 1))
                goto next;
            {
                zero = 0;
                dma->src = &zero;
                dma->dst = (void *)0x0600f000;
                dma->cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x400;
                (void)dma->cnt;

                dma->src = &zero;
                dma->dst = (void *)0x02020000;
                dma->cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x8000;
                (void)dma->cnt;

                sub_08011984(sBlitRecords_6d40, (u8)(i * 3));
                sub_0800EE94(2);

                *(vu16 *)0x04000050 = 0x1744;
                *(vu16 *)0x04000052 = 0x020e;

                base[0x2b] |= (u8)(1 << i);
                gIwram_6480._field_02 = (u8)i;
            }
        next:
            i = (u16)(i + 1);
        } while (i <= 1);
    }

    if ((s8)sub_08011AA4(sBlitRecords_6d40) != 0)
        result = 1;

    return result;
}

void sub_08011F3C(const struct BlitRecord *records, u8 idx)
{
    const struct BlitRecord *rec;
    const struct BlitRecord *next;
    const u16 *src;
    u16 *dst;
    s32 oy;
    s32 ix;
    s32 jx;

    gIwram_53A0[0] = 0;
    gIwram_6480.state = 1;
    gIwram_6480._field_09 = 1;

    rec = &records[idx];
    next = rec + 1;

    gIwram_6480._field_36 = next->tilesPerCol;
    gIwram_6480._field_34 = next->_field_00;
    gIwram_6480.src = next->src;
    gIwram_6480.mode = 2;
    gIwram_6480.dst = (u16 *)0x02020000;
    gIwram_6480._field_0b = next->_field_0c;
    gIwram_6480.width = next->_field_04;
    gIwram_6480.height = next->_field_06;
    gIwram_6480._field_2c = next->_field_04;
    gIwram_6480._field_2e = next->_field_06;

    dst = (u16 *)(0x02020000 + 2 * (next->_field_00 + gIwram_60A0.rowStride * next->tilesPerCol));
    gIwram_6480.dst = dst;

    if (next->_field_06 != 0) {
        for (oy = 0; (u16)oy < gIwram_6480.height; oy++) {
            src = gIwram_6480.src;
            for (ix = 0; (u16)ix < gIwram_6540.height; ix++) {

                    *dst++ = *src++;
                dst += gIwram_60A0.rowStride - gIwram_6540.width;
            }

    src = rec[2].src;
    if (gIwram_6540.height != 0) {
        for (oy = 0; (u16)oy < gIwram_6540.height; oy++) {

}
