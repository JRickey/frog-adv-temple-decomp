0x0802a63c..0x0802a8fc (704 bytes). Entity spawn/age dispatcher for the
section-spawn grid. Reads section index `*(s8*)(gEntities + 0x6aa)`
(= `gEntities[30].field_1A`), looks up an 8-byte `SpawnGridSubtable` entry via
`sSpriteGridSubtable` (ROM 0x083067d4) to get a 12-byte `SpawnGridEntry` slice
pointer, and loops `i = 0..7` over `gEntities[i+2]`, spawning/aging entities
per two u16 bitmasks: maskLo at `gEntities[30].field_08` (base+0x698,
"spawned") and maskHi at `gEntities[30].field_0A` (base+0x69a == absolute
0x03003dba, "done").

## KEY NEW FINDING (round 35) — the loop chain IS reproducible

`<that .s>: CFLAGS += -ffixed-r6` REPRODUCES THE EXACT baserom loop high-reg
chain: constant `1` → r8, counter `i<<24` → r9, slice → sl, tick → spilled
[sp,#24]. Confirmed by disasm: `movs r3,#1; mov r8,r3` then `mov r9,r0`
(counter) — byte-identical hoist shape to baserom 0x2a6fc/0x2a70e.

BUT `-ffixed-r6` is NOT a valid fix: the baserom USES r6 as a heavily-reused
SCRATCH register (maskHi address `adds r6,r4,r1`; the `1<<i` shift base; many
loop-body temporaries) AND saves it in the prologue `push {r4,r5,r6,r7,lr}`.
`-ffixed-r6` bans r6 → prologue becomes `push {r4,r5,r7,lr}` (r6 dropped) and
every r6-scratch site reshuffles → 506/256/612, with 219 residual loop diffs
from the scratch ripple. So it proves REACHABILITY (the chain is not exotic),
not a match.

**Interpretation:** the divergence is PURELY that the baserom loop carries ONE
MORE callee-saved-live value than agbcc naturally keeps, which spills `1` into
r8 and chain-shifts counter→r9, slice→sl, tick→spill. `-ffixed-r6` fakes that
pressure by removing a low reg. The real fix is a SOURCE shape that adds the
4th callee-saved-live value (base held live in r7 across the WHOLE loop, used
for the entity address like the baserom) WITHOUT the size cost — OR the permuter.

## The base-liveness tension (the actual wall)

The baserom keeps `base`(r7) LIVE across the entire loop because it recomputes
the entity address `base + 0x38*(i+2)` (and `base + 0x70 + 0x38*i` for the
+0x30 fields) INLINE at every branch from r7+counter. That keeps 4 values
callee-saved-live (base, 1, counter, slice) → forces high-reg + tick spill.

- **cache `e` once at loop top** (`e = (struct Entity*)(base + (i+2)*0x38);`):
  base DIES right after → r7 freed → counter steals r7, `1` rematerialized.
  byte_diff 490, size 592 (TOO SMALL, no high regs). The most readable + best
  byte_diff, but wrong coloring.
- **inline `e` from base at every use**: base STAYS live (good, 4th value) but
  agbcc emits a 6-insn address recompute (`...; adds r0,#0x70`) where the
  baserom uses a shared 5-insn `(i+2)*0x38` form → size balloons to 660-680.
  diff_count drops to 195 (best STRUCTURE) but byte_diff 584+.

So: cache-e wins byte_diff but loses the coloring; inline-e wins the coloring +
structure but loses size to address-recompute bloat. The baserom does inline-e
WITH a tighter shared recompute that agbcc won't reproduce from any source shape
tried.

## Confirmed-correct structural facts (KEEP these)

1. `struct SpawnGridEntry { s16 x; s16 y; u8 col; u8 dy; u8 mode; s8 delay;
   u8 flags; u8 _pad09[3]; }` — 12-byte stride. `struct SpawnGridSubtable
   { u32 count; const SpawnGridEntry *slice; }[8]` over sSpriteGridSubtable.
2. `slice = sSpriteGridSubtable[*(s8*)(base+0x6aa)].slice`.
3. **maskHi inside the loop is read/written via the ABSOLUTE pointer**
   `(s16*)0x03003dba` / `(u16*)0x03003dba` (its own pool literal), NOT
   base+0x69a. This is what makes SIZE = 608 (base-relative gives 592). The
   SETUP maskHi (reset block) uses base+0x69a.
4. maskLo (`base+0x698`) is read SIGNED `(s16*)` (ldrsh) and shifted arithmetic
   `>> i`; same for maskHi. Both `& 1`.
5. delay check is UNSIGNED: `if ((u32)(tick - *(u32*)(base+0x6b0)) <=
   (u32)(s8)slice[i].delay)` — `bls`, delay is `ldrsb` then unsigned-compared.
6. Entity_Init signature (from src/engine/sub_0800a710.c) is
   `(struct Entity*, u8, u16 x, u16 y, u8, u16, u8, u8, u8, u16)` — x/y are u16.
   Args: `(e, 0x51, slice[i].x, slice[i].y, 3, (i%8)*16 + 0x41, 1, 3, 2, 16)`.
7. base+0x6b0 (lastTick) is written ONLY in the reset (maskHi==0xff) block,
   twice (before+after sub_08000764 section randomization). Main path leaves it.
8. Reset block: `gIwram_6110.flags2--`, zero maskLo+maskHi, lastTick=_unk00,
   then gateByte 2→rand(5)+3, 1→rand(5)+3, else→rand(3); then lastTick=_unk00,
   zero maskLo+maskHi again, `if (flags2==0) return;`.
9. e callee `Entity_Init`/0x0800a710 is ALREADY DECOMPILED (decomp_brief's
   "UNPEELED ✗" is a FALSE POSITIVE — 0x0800a710 lives in src/engine/
   sub_0800a710.c with a real `.globl Entity_Init`). No peel needed.

#include "game.h"
#include "iwram.h"
#include "types.h"

extern u8 sub_08000764(u8 range);
extern void sub_08020C78(u32 sound);
extern void sub_0800696C(struct IwramAt6110 *control, s32 slot);
extern void sub_080059C4(struct Entity *e);
extern void sub_08020F3C(u8 slot);
extern void Entity_Init(struct Entity *p, u8 a, u16 x, u16 y, u8 actorId, u16 e, u8 f, u8 g, u8 state,
                        u16 initFlags);

struct SpawnGridEntry {
    s16 x;
    s16 y;
    u8 col;
    u8 dy;
    u8 mode;
    s8 delay;
    u8 flags;
    u8 _pad09[3];
};

struct SpawnGridSubtable {
    u32 count;
    const struct SpawnGridEntry *slice;
};

extern const struct SpawnGridSubtable sSpriteGridSubtable[8];

void sub_0802A63C(void)
{
    register u8 *base asm("r7");
    const struct SpawnGridEntry *slice;
    u32 tick;
    s8 i;
    struct Entity *e;

    base = (u8 *)0x03003720;
    slice = sSpriteGridSubtable[*(s8 *)(base + 0x6aa)].slice;

    if (*(u16 *)(base + 0x69a) == 0xff) {
        gIwram_6110.flags2--;
        *(u16 *)(base + 0x698) = 0;
        *(u16 *)(base + 0x69a) = 0;
        *(u32 *)(base + 0x6b0) = gGameStuff._unk00;
        if (gIwram_6110.gateByte == 2)
            *(s8 *)(base + 0x6aa) = sub_08000764(5) + 3;
        else if (gIwram_6110.gateByte == 1)
            *(s8 *)(base + 0x6aa) = sub_08000764(5) + 3;
        else
            *(s8 *)(base + 0x6aa) = sub_08000764(3);

        *(u32 *)(base + 0x6b0) = gGameStuff._unk00;
        *(u16 *)(base + 0x698) = 0;
        *(u16 *)(base + 0x69a) = 0;
        if (gIwram_6110.flags2 == 0)
            return;
    }

    tick = gGameStuff._unk00;

    for (i = 0; i <= 7; i++) {
        e = &gEntities[i + 2];

        if ((*(s16 *)0x03003dba >> i) & 1) {
            e->status |= 8;
            continue;
        }

        if ((*(s16 *)(base + 0x698) >> i) & 1)
            goto age;

        if (tick - *(u32 *)(base + 0x6b0) <= (u32)slice[i].delay) {
            e->status |= 8;
            continue;
        }

        Entity_Init(e, 0x51, slice[i].x, slice[i].y, 3, (i % 8) * 16 + 0x41, 1, 3, 2, 16);

        if (slice[i].flags & 1)
            sub_08020C78(0x3d);

        if (gIwram_6110.gateByte == 0) {
            e->field_32 = slice[i].mode;
            e->field_31 = slice[i].dy;
        } else {
            e->field_31 = slice[i].dy - 1;
            e->field_32 = slice[i].mode;
        }
        e->field_30 = slice[i].col;

        *(u16 *)(base + 0x698) |= 1 << i;
        sub_0800696C(&gIwram_6110, i + 2);
        sub_080059C4(e);
        continue;

    age:
        if (e->status & 8)
            continue;
        if ((s16)e->y > 178) {
            *(u16 *)0x03003dba |= 1 << i;
            e->status |= 8;
            continue;
        }
        if (e->status & 0x8000)
            e->status = (e->status & 0x7fff) | 2;
        sub_08020F3C((u8)(i + 2));
        sub_080059C4(e);
    }
}
