#define BLIT_BANK_HI ((u16 *)0x02010000)

typedef struct BlitEntry {
    u16 dstX;
    u16 dstY;
    u16 cols;
    u16 rows;
    u8 flags;
    u8 _pad09[7];
    const u16 *const *srcTable;
    u8 _pad14[4];
} BlitEntry;

typedef struct BlitState {
    u8 _pad00[12];
    const u16 *src;
    u8 _pad10[4];
    u16 *dst;
} BlitState;

typedef struct SceneScroll {
    u8 _pad00[26];
    u16 stride;
} SceneScroll;

#define gSceneScroll (*(SceneScroll *)0x030060A0)
