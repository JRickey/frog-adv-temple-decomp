#include "macros.h"
#include "types.h"
#include "iwram.h"
#include "gba/dma.h"

#define TILE_BUFFER_0 ((u16 *)0x02000000)
#define TILE_BUFFER_1 ((u16 *)0x02010000)
#define TILE_BUFFER_2 ((u16 *)0x02020000)

struct SceneScrollState_15C24 {
    u8 _pad00[26];
    u16 tileWidth;
};

#define SCENE_SCROLL_STATE_15C24 ((struct SceneScrollState_15C24 *)0x030060A0)

void sub_08015C24(u8 bank, u16 dstX, u16 dstY, u8 cols, u8 rows, const u16 *src)
{
    u16 *dst;
    struct SceneScrollState_15C24 *state;
    register s32 x asm("r1");
    register s32 y asm("r0");
    register s32 xScratch asm("r0");
    register u32 row asm("r0");

    switch (bank) {
    case 0:
        dst = TILE_BUFFER_0;
        break;
    case 1:
        dst = TILE_BUFFER_1;
        break;
    case 2:
        dst = TILE_BUFFER_2;
        break;
    }

    xScratch = dstX << 16;
    x = xScratch >> 16;
    y = (s16)dstY;
    {
        register struct SceneScrollState_15C24 *st asm("r2") = SCENE_SCROLL_STATE_15C24;
        y *= st->tileWidth;
        dst += y + x;
        state = st;
    }

    row = 0;
    if (row < rows) {
        struct SceneScrollState_15C24 *loopState = state;
        do {
            register u8 nextRow asm("r1");
            u8 col = 0;
            nextRow = row + 1;
            if (col < cols) {
                do {
                    *dst++ = *src++;
                    col++;
                } while (col < cols);
            }
            {
                register u16 stride asm("r2") = loopState->tileWidth;
                row = stride;
                row -= cols;
                dst += row;
            }
            row = (u8)nextRow;
        } while (row < rows);
    }
}
