## What the function does (semantics — confirmed correct)

`void sub_08010DD8(struct DrawRecord *records, u32 n /* (u8) */, u32 commit)`

- A = 0x03006110 (control struct). `cur = A[0x2a]`, `prev = A[0x2b]` are bitmasks.
  `bitA = (cur >> n) & 1`. Continue into the body iff `(bitA && bitA != (prev>>n&1))`
  (a rising-edge on bit n) OR `commit == 0`; else early-return.
- B = 0x08307EAC (ROM), stride-20 scene descriptors. `stream = *(u32*)(B + 12 +
  (gGameStuff.pendingMode - 1)*20)` — field +0 of the (mode-1)th record is a byte
  stream base. NOTE: the `+12` is a RUNTIME add and `(mode-1)` is computed
  explicitly in the baserom (agbcc otherwise CSE-folds `(mode-1)*20 + 12` →
  `mode*20 - 8`; defeated here with `sceneTable=0x08307EAC; asm("":"+r"); sceneTable+=12;`).
- C = 0x030063F0 (16-byte byte record). `sum = C[n] + ((u8*)stream)[n*2]`.
  `rec = (DrawRecord*)((u8*)records + sum*24)` — stride-24 record.
- D = 0x03006540 (blit-state block). `flagBit = (rec->flags >> 4) & 1`.
  **D[8] = flagBit; D[24] = flagBit ? 0x02010000 : 0x02000000** (EWRAM bank select —
  this was the one logic bug an earlier mental trace got wrong: the !flagBit branch
  sets D[24]=0x02000000 via `movs r1,#0x80; lsls #18`, NOT 0).
- F = 0x030035E0: `*(u16*)0x030035F0 = 1` (F[16]=1).
- D[0] = C[n] + 1; if commit: D[2] += 1<<n.
- D[11]=rec->_unk0C; D[48]=D[44]=rec->_unk04; D[50]=D[46]=rec->_unk06 (cache the two
  halfwords in locals `w,h` — without caching agbcc re-reads rec for the 44/46 stores).
- `switch (rec->flags & 15)` cases 1/2/3/4 (1 and 3 share a body), default no-op.
  cols = 0x030060A0, cols[13] (= halfword at +26) is a stride/width. Cases adjust
  D[24] (src) and D[16] (dst). See the Best-effort C for the exact arithmetic
  (matches the baserom case bodies instruction-for-instruction modulo coloring).
- Tail: if commit: A[0x2b] = A[0x2a] (latch prev = cur).

#include "game.h"
#include "macros.h"
#include "types.h"

struct DrawRecord {
    u16 _unk00;
    u16 _unk02;
    u16 _unk04;
    u16 _unk06;
    u8 flags;
    u8 _pad09[3];
    u8 _unk0C;
    u8 _pad0D[3];
    u32 *_unk10;
};

void sub_08010DD8(struct DrawRecord *records, u32 n, u32 commit)
{
    register struct DrawRecord *recs asm("r5") = records;
    u32 commitFlag = commit;
    register u32 idx asm("r4") = (u8)n;
    register u8 *streamBytes asm("r6");
    u8 *ctrl = (u8 *)0x03006110;
    register u8 *ctrlSave asm("ip");
    register struct DrawRecord *rec asm("r3");
    u8 *blit;
    u32 *ewram;
    u16 *cols;
    u32 stream;
    u32 sceneIdx;
    u8 *sceneTable;
    u32 bitA;

    bitA = (ctrl[0x2a] >> idx) & 1;
    ctrlSave = ctrl;
    if (bitA != 0 && bitA != ((ctrl[0x2b] >> idx) & 1)) {
        /* rising edge on bit idx — fall through */
    } else if (commitFlag != 0) {
        return;
    }

    /* asm-barrier + runtime += 12 keep agbcc from folding (mode-1)*20+12 into
       one literal; sceneTable referenced before gGameStuff fixes pool order. */
    sceneTable = (u8 *)0x08307EAC;
    asm("" : "+r"(sceneTable));
    sceneTable += 12;
    sceneIdx = gGameStuff.pendingMode - 1;
    stream = *(u32 *)(sceneTable + sceneIdx * 20);
    streamBytes = (u8 *)0x030063F0;
    rec = (struct DrawRecord *)((u8 *)recs +
                                (streamBytes[idx] + ((u8 *)stream)[idx * 2]) * 24);

    blit = (u8 *)0x03006540;
    if ((rec->flags >> 4) & 1) {
        blit[8] = 1;
        ewram = (u32 *)0x02010000;
    } else {
        blit[8] = 0;
        ewram = (u32 *)0x02000000;
    }
    *(u32 **)(blit + 24) = ewram;

    *(u16 *)0x030035F0 = 1;

    blit[0] = streamBytes[idx] + 1;
    if (commitFlag != 0) {
        blit[2] += 1 << idx;
    }

    blit[11] = rec->_unk0C;
    {
        u16 w = rec->_unk04;
        u16 h = rec->_unk06;
        *(u16 *)(blit + 48) = w;
        *(u16 *)(blit + 50) = h;
        *(u16 *)(blit + 44) = w;
        *(u16 *)(blit + 46) = h;
    }

    cols = (u16 *)0x030060A0;
    switch (rec->flags & 15) {
    case 2:
        *(u32 *)(blit + 24) += 2 * (rec->_unk00 + cols[13] * (rec->_unk06 + rec->_unk02 - 1));
        *(u32 *)(blit + 16) = *rec->_unk10 + 2 * ((rec->_unk06 - 1) * rec->_unk04);
        break;
    case 1:
    case 3:
        *(u32 *)(blit + 24) += 2 * (rec->_unk00 + cols[13] * rec->_unk02);
        *(u32 *)(blit + 16) = *rec->_unk10;
        break;
    case 4:
        *(u32 *)(blit + 24) += 2 * (rec->_unk00 + cols[13] * rec->_unk02);
        *(u32 *)(blit + 24) += rec->_unk04 * 2 - 2;
        *(u32 *)(blit + 16) = *rec->_unk10;
        *(u32 *)(blit + 16) += rec->_unk04 * 2 - 2;
        break;
    }

    if (commitFlag != 0) {
        ctrlSave[0x2b] = ctrlSave[0x2a];
    }
}
