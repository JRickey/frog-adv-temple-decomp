   * `register u32 tick asm("r7")` — keeps tick in r7 (else agbcc spills it to a
     slot, growing the frame to 60 and cascading every sp-relative access).
   * declaration order `..., tagByte, i, ...` so tag homes to sp#16 and first to

     union AnimRect rect2;`. agbcc assigns named-local slots in declaration order,

     the target. (Declaring idShift before/after the rects puts it at sp#28/sp#52.)

3. The mask INTERLEAVING (maskHi/maskLo materialized at first use INSIDE the rect0

struct AnimCell {
    u16 y;        /* +0 */
    u16 x;        /* +2 */
    u16 height;   /* +4 */
    u16 width;    /* +6 */
    u8 _pad08[12];
    u32 stamp;    /* +20: last-advance tick */
    u8 _pad18;
    u8 counter;   /* +25 */
    u8 flags;     /* +26: bit 2 = state==2 */
    u8 state;     /* +27 */
    u8 _pad1C[4];
    const s16 *records; /* +32 */
};

struct ScriptAnim {
    u8 _b0;
    u8 _pad01[3];
    u8 _b4;
    u8 _b5;
    s8 _b6;
    s8 _b7;
    s8 _b8;
    s8 _b9;
    u8 _pad0A[2];
};

union AnimRect {
    struct {
        u32 point;
        u32 size;
    } w;
};

extern s32 sub_0800CED0(void *ctx, u32 point, u32 size, s32 c, s32 d);
extern void sub_08020DC4(u8 idx);
extern void sub_08020C78(u32 sound);

/* sEntityScriptIndexExt re-typed for the signed-byte threshold fields this
 * consumer reads from each 12-byte record. */
#define sEntityScriptAnim ((const struct ScriptAnim *)sEntityScriptIndexExt)

/* Center rect {x-width/2, y-height/2-1, height, width}; the two s16 halves are
 * merged into point.w / size.w with the 0xffff0000 / 0x0000ffff clear masks,
 * which agbcc holds loop-invariant in the high regs r9 / r8 (cf. the matched
 * sub_08007138 center-rect packer). */
#define PACK_RECT(rect, c)                                                                                              \
    do {                                                                                                               \
        r2v = (c)->width;                                                                                              \
        r3v = r2v << 16;                                                                                               \
        r0v = r3v >> 17;                                                                                               \
        r1v = (c)->x;                                                                                                  \
        r0v = r1v - r0v;                                                                                               \
        r0v <<= 16;                                                                                                    \
        r0v >>= 16;                                                                                                    \
        r1v = maskHi;                                                                                                  \
        r1v &= (rect).w.point;                                                                                         \
        r1v |= r0v;                                                                                                    \
        r2v = (c)->height;                                                                                             \
        r0v = r2v << 16;                                                                                               \
        r2v = r0v >> 16;                                                                                               \
        r0v >>= 17;                                                                                                    \
        r4v = (c)->y;                                                                                                  \
        r0v = r4v - r0v;                                                                                               \
        r0v -= 1;                                                                                                      \
        r0v <<= 16;                                                                                                    \
        r4v = maskLo;                                                                                                  \
        r1v &= r4v;                                                                                                    \
        (rect).w.point = r1v | r0v;                                                                                    \
        r0v = maskHi;                                                                                                  \
        r0v &= (rect).w.size;                                                                                          \
        r0v |= r2v;                                                                                                    \
        r0v &= r4v;                                                                                                    \
        (rect).w.size = r0v | r3v;                                                                                     \
    } while (0)

void sub_0800736C(s8 partId, s32 cells, s32 first, s32 last, s8 tag)
{
    register u32 maskHi asm("r9");
    register u32 maskLo asm("r8");
    register s8 scriptId asm("sl");
    register u32 tick asm("r7");
    struct AnimCell *cell;
    u8 tagByte;
    s32 i;
    u8 acted;
    s8 idLive;
    u32 r0v, r1v, r2v, r4v;
    register u32 r3v asm("r3");
    union AnimRect rect0;
    s32 idShift;
    union AnimRect rect1;
    union AnimRect rect2;

    tagByte = tag;
    tick = gGameStuff._unk00;
    acted = 0;
    i = first;
    if (i > last)
        return;

    cell = (struct AnimCell *)(i * 36 + cells);
    idShift = partId << 24;
    scriptId = idShift >> 24;

    do {
        if ((struct AnimCell *)gEntities == NULL || cell == NULL)
            goto next;

        maskHi = 0xFFFF0000;
        r2v = cell->width;
        r3v = r2v << 16;
        r0v = r3v >> 17;
        r1v = cell->x;
        r0v = r1v - r0v;
        r0v <<= 16;
        r0v >>= 16;
        r1v = maskHi;
        r1v &= rect0.w.point;
        r1v |= r0v;
        r2v = cell->height;
        r0v = r2v << 16;
        r2v = r0v >> 16;
        r0v >>= 17;
        r4v = cell->y;
        r0v = r4v - r0v;
        r0v -= 1;
        r0v <<= 16;
        maskLo = 0x0000FFFF;
        r4v = maskLo;
        r1v &= r4v;
        rect0.w.point = r1v | r0v;
        r0v = maskHi;
        r0v &= rect0.w.size;
        r0v |= r2v;
        r0v &= r4v;
        rect0.w.size = r0v | r3v;

        if (sub_0800CED0(gEntities, rect0.w.point, rect0.w.size, 0x104, 0xb4) == 0)
            goto next;

        switch (cell->state) {
        case 0:
            idLive = idShift >> 24;
            if (tick - cell->stamp > (u32)(s8)sEntityScriptAnim[idLive]._b7) {
                cell->state = 1;
                cell->stamp = tick;
                acted = 1;

                PACK_RECT(rect1, cell);
                if (sub_0800CED0(gEntities, rect1.w.point, rect1.w.size, 0x48, 0x48) != 0 && idLive == 0)
                    sub_08020DC4(7);
            }
            break;
        case 1:
            if (tick - cell->stamp > (u32)sEntityScriptAnim[scriptId]._b8) {
                if (cell->counter >= sEntityScriptAnim[scriptId]._b6 - 1) {
                    cell->state = 2;
                    cell->counter = sEntityScriptAnim[scriptId]._b6 - 1;
                } else {
                    cell->counter = cell->counter + 1;
                }
                cell->stamp = tick;
                acted = 1;
            }
            break;
        case 2:
            idLive = idShift >> 24;
            if (tick - cell->stamp > (u32)(s8)sEntityScriptAnim[idLive]._b9) {
                cell->state = 3;
                cell->stamp = tick;
                acted = 1;

                PACK_RECT(rect2, cell);
                if (sub_0800CED0(gEntities, rect2.w.point, rect2.w.size, 0x48, 0x48) != 0 && idLive == 1)
                    sub_08020C78(0x56);
            }
            break;
        case 3:
            if (tick - cell->stamp > (u32)sEntityScriptAnim[scriptId]._b8) {
                u8 c = cell->counter;
                if (c != 0) {
                    cell->counter = c - 1;
                } else {
                    cell->state = 0;
                    cell->counter = 0;
                }
                cell->stamp = tick;
                acted = 1;
            }
            break;
        }

        if (cell->state == 2)
            cell->flags |= 4;
        else
            cell->flags &= ~4;

        if (acted == 1) {
            const s16 *rec = cell->records;
            sub_080112C0(sEntityScriptAnim[scriptId]._b0, sEntityScriptAnim[scriptId]._b5,
                         sEntityScriptAnim[scriptId]._b4, (u16)(rec[0] * 3), (u16)(rec[1] * 3), tagByte,
                         cell->counter);
        }

    next:
        cell = (struct AnimCell *)((s32)cell + 36);
        i++;
    } while (i <= last);
}
