## NEW findings this round (advance over the prior note)

A. **POINT-WORD two-store keeps slots; SIZE-WORD two-store BREAKS them.**
   The target stores rect.point TWICE (after .x merge, after .y merge) and
   rect.size TWICE (after height merge, after width merge). Writing point as a
   two-store sequence:
       r1v = maskHi; r1v &= rect.w.point; r1v |= pointX; rect.w.point = r1v;  // store1
       r1v &= maskLo; r1v |= (pointY<<16); rect.w.point = r1v;                // store2
   KEEPS the prologue/slots correct (i@20, acted@24, idShift@36) and matches the
   point word -> byte_diff 623, the new best clean base. But applying the SAME
   two-store to the SIZE word regresses: i spills LATE to sp#52, acted->sp#20,
   idShift->sp#32 (frame layout shifts), byte_diff 666-689. This is a reload
   spill-ordering effect (gcc_arm/reload1.c / local-alloc.c): the extra size
   `str` changes which scalars get a home slot at decl-time vs spilled during
   reload. NO decl-order permutation of idShift/acted/i fixes it (tried: idShift
   between rects #32, idShift last #48, acted last #48, volatile i — all #52).
   => Single-store (622, slots right, packing wrong) and full two-store (689,
   packing right, slots wrong) are two distinct sharp minima. Point-two-store
   (623) is the new pareto point: slots right + point word right, only the
   SIZE word still single-store.

B. **maskHi hoist via loop-invariant motion (gcc_arm/loop.c).** The target
   RE-LOADS maskHi (`ldr =0xffff0000; mov r9`) inside rect0 EVERY iteration.
   Our `maskHi = 0xFFFF0000` at the loop top (or at first use after the guards)
   is a loop-invariant constant -> agbcc hoists the load before the loop, and
   the point.x merge becomes `ldr point; mov r2,r9; ands r1,r2` instead of the
   target's `mov r1,r9; ldr point(r4); ands r1,r4`. gcc 2.x has no
   `-fno-move-loop-invariants`; could not defeat the hoist from pure C.

C. **PackedRect (s16 fields) does NOT give masked-RMW on the stack.** The
   matched sister sub_080076A4 uses `struct PackedRect { s16 x,y,height,width; }`
   passed by value and produces the masked-RMW packing — but ONLY because there
   the rect lives in REGISTERS (r6/r7). On the STACK, s16 field stores lower to
   aligned `strh`, not the 32-bit masked RMW the target uses. So PackedRect is a
   dead end here; the union-of-u32 form (below) is required.

D. **TU-wide CFLAGS are BLOCKED.** This TU also contains the matched siblings
   sub_08007228 (NAKED) and sub_080072E0 (true match). `-fno-gcse`,
   `-fno-schedule-insns2`, and the newer-agbcc compiler swap each make
   sub_080072E0 NONMATCHING (byte_diff 3-5) while barely moving sub_0800736C.
   So no TU-wide flag lever is available.

E. **Cross-function regression while non-matching (make check blocker).** With a
   NON-MATCHING sub_0800736C appended, sub_080072E0 *also* goes byte_diff 3
   (its own `lsrs`/register coloring shifts) PURELY from the new function being
   in the same TU — even with zero source changes before sub_080072E0. The
   baserom proves all three match together, so this is a symptom of
   sub_0800736C being non-matching (TU constant-pool/layout shift), not an
   independent bug. It does mean a near-miss sub_0800736C cannot be shipped
   under NON_MATCHING in this TU without breaking the sibling.

#include "iwram.h"
#include "macros.h"
#include "types.h"

/* (sub_08007228 + sub_080072E0 unchanged — see the live file) */

struct AnimCell {
    u16 y;        /* +0 */
    u16 x;        /* +2 */
    u16 height;   /* +4 */
    u16 width;    /* +6 */
    u8 _pad08[12];
    u32 stamp;    /* +20 */
    u8 _pad18;
    u8 counter;   /* +25 */
    u8 flags;     /* +26: bit 2 = state==2 */
    u8 state;     /* +27 */
    u8 _pad1C[4];
    const s16 *records; /* +32 */
};

struct ScriptAnim {
    u8 _b0;
    u8 _pad01[3];
    u8 _b4;
    u8 _b5;
    s8 _b6;
    s8 _b7;
    s8 _b8;
    s8 _b9;
    u8 _pad0A[2];
};

union AnimRect {
    struct {
        u32 point;
        u32 size;
    } w;
};

extern s32 sub_0800CED0(void *ctx, u32 point, u32 size, s32 c, s32 d);
extern void sub_08020DC4(u8 idx);
extern void sub_08020C78(u32 sound);
extern void sub_080112C0(u8 a, u8 b, u8 c, u16 d, u16 e, u8 tag, u8 counter);
extern const struct EntScriptExt sEntityScriptIndexExt[];

#define sEntityScriptAnim ((const struct ScriptAnim *)sEntityScriptIndexExt)

/* point: TWO stores (slot-safe). size: ONE store (two stores break the slots). */
#define PACK_RECT(rect, c)                                                     \
    do {                                                                       \
        r2v = (c)->width;                                                      \
        r3v = r2v << 16;                                                       \
        r0v = r3v >> 17;                                                       \
        r1v = (c)->x;                                                          \
        r0v = r1v - r0v;                                                       \
        r0v <<= 16;                                                            \
        r0v >>= 16;                                                            \
        r1v = maskHi;                                                          \
        r1v &= (rect).w.point;                                                 \
        r1v |= r0v;                                                            \
        (rect).w.point = r1v;                                                  \
        r2v = (c)->height;                                                     \
        r0v = r2v << 16;                                                       \
        r2v = r0v >> 16;                                                       \
        r0v >>= 17;                                                            \
        r4v = (c)->y;                                                          \
        r0v = r4v - r0v;                                                       \
        r0v -= 1;                                                              \
        r0v <<= 16;                                                            \
        r4v = maskLo;                                                          \
        r1v &= r4v;                                                            \
        r1v |= r0v;                                                            \
        (rect).w.point = r1v;                                                  \
        r0v = maskHi;                                                          \
        r0v &= (rect).w.size;                                                  \
        r0v |= r2v;                                                            \
        r0v &= r4v;                                                            \
        (rect).w.size = r0v | r3v;                                             \
    } while (0)

void sub_0800736C(s8 partId, s32 cells, s32 first, s32 last, s8 tag)
{
    register u32 maskHi asm("r9");
    register u32 maskLo asm("r8");
    register s8 scriptId asm("sl");
    register u32 tick asm("r7");
    struct AnimCell *cell;
    u8 tagByte;
    s32 i;
    u8 acted;
    s8 idLive;
    u32 r0v, r1v, r2v, r4v;
    register u32 r3v asm("r3");
    union AnimRect rect0;
    s32 idShift;
    union AnimRect rect1;
    union AnimRect rect2;

    tagByte = tag;
    tick = *(volatile u32 *)0x03005330; /* gGameStuff._unk00; game.h include makes 072E0 nonmatch */
    acted = 0;
    i = first;
    if (i > last)
        return;

    cell = (struct AnimCell *)(i * 36 + cells);
    idShift = partId << 24;
    scriptId = idShift >> 24;

    do {
        if ((struct AnimCell *)gEntities == NULL || cell == NULL)
            goto next;

        maskHi = 0xFFFF0000;
        r2v = cell->width;
        r3v = r2v << 16;
        r0v = r3v >> 17;
        r1v = cell->x;
        r0v = r1v - r0v;
        r0v <<= 16;
        r0v >>= 16;
        r1v = maskHi;
        r1v &= rect0.w.point;
        r1v |= r0v;
        rect0.w.point = r1v;
        r2v = cell->height;
        r0v = r2v << 16;
        r2v = r0v >> 16;
        r0v >>= 17;
        r4v = cell->y;
        r0v = r4v - r0v;
        r0v -= 1;
        r0v <<= 16;
        maskLo = 0x0000FFFF;
        r4v = maskLo;
        r1v &= r4v;
        r1v |= r0v;
        rect0.w.point = r1v;
        r0v = maskHi;
        r0v &= rect0.w.size;
        r0v |= r2v;
        r0v &= r4v;
        rect0.w.size = r0v | r3v;

        if (sub_0800CED0(gEntities, rect0.w.point, rect0.w.size, 0x104, 0xb4) == 0)
            goto next;

        switch (cell->state) {
        case 0:
            idLive = idShift >> 24;
            if (tick - cell->stamp > (u32)(s8)sEntityScriptAnim[idLive]._b7) {
                cell->state = 1;
                cell->stamp = tick;
                acted = 1;

                PACK_RECT(rect1, cell);
                if (sub_0800CED0(gEntities, rect1.w.point, rect1.w.size, 0x48, 0x48) != 0 && idLive == 0)
                    sub_08020DC4(7);
            }
            break;
        case 1:
            if (tick - cell->stamp > (u32)sEntityScriptAnim[scriptId]._b8) {
                if (cell->counter >= sEntityScriptAnim[scriptId]._b6 - 1) {
                    cell->state = 2;
                    cell->counter = sEntityScriptAnim[scriptId]._b6 - 1;
                } else {
                    cell->counter = cell->counter + 1;
                }
                cell->stamp = tick;
                acted = 1;
            }
            break;
        case 2:
            idLive = idShift >> 24;
            if (tick - cell->stamp > (u32)(s8)sEntityScriptAnim[idLive]._b9) {
                cell->state = 3;
                cell->stamp = tick;
                acted = 1;

                PACK_RECT(rect2, cell);
                if (sub_0800CED0(gEntities, rect2.w.point, rect2.w.size, 0x48, 0x48) != 0 && idLive == 1)
                    sub_08020C78(0x56);
            }
            break;
        case 3:
            if (tick - cell->stamp > (u32)sEntityScriptAnim[scriptId]._b8) {
                u8 c = cell->counter;
                if (c != 0) {
                    cell->counter = c - 1;
                } else {
                    cell->state = 0;
                    cell->counter = 0;
                }
                cell->stamp = tick;
                acted = 1;
            }
            break;
        }

        if (cell->state == 2)
            cell->flags |= 4;
        else
            cell->flags &= ~4;

        if (acted == 1) {
            const s16 *rec = cell->records;
            sub_080112C0(sEntityScriptAnim[scriptId]._b0, sEntityScriptAnim[scriptId]._b5,
                         sEntityScriptAnim[scriptId]._b4, (u16)(rec[0] * 3), (u16)(rec[1] * 3), tagByte,
                         cell->counter);
        }

    next:
        cell = (struct AnimCell *)((s32)cell + 36);
        i++;
    } while (i <= last);
}
