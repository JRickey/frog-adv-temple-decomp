# Deferred analysis: sub_0800E85C (0x0800e85c, 704 bytes)

Verdict remains `ATTEMPT_MATCH`, not `STRONG_UNMATCHABLE`.
`classify_unmatchable.py` reports only the advisory high-reg case:
`r8`/`r9` live across a `bl`. Do not ship NAKED for this target.

## Drift

The function is structurally understood: `count` selects how many of the
three character-sprite/BG layers to load. The loop indexes
`sCharacterSpriteFrames[gGameStuff.pendingMode * 3 + pose]`, writes
width/height into the layer-control area at `0x030060A0 + pose * 0x20`,
then dispatches pose-specific DMA/LZ77 loads. The tail uploads the palette,
sets `REG_DISPCNT`, toggles BG layers via `sub_0800EE94`/`sub_0800EE34`,
disables HBlank IRQ state, and writes `gIwram_3608._pad4 = 1`,
`gIwram_3608._field_5 = 0`.

Prerequisite callee status changed this round: `sub_0802D568` is now a real
peeled Thumb symbol in commit `Peel sub_0802D568`, so no local `.thumb_set`
workaround is needed in the C attempt.

Best this round: `byte_diff 449`, `diff_count 193`, built with default
`OLD_AGBCC_BIN`, with pins for `game=r5`, `frames=r8`, `tilesA=sl`,
`cnt=r9`. The root mismatch is still the same allocator live-range plateau:
the built code emits `adds r4, r3, #0` immediately after the zero-count check,
before the loop-invariant frame/game loads. The baserom emits:

```asm
ldr  r0, =sCharacterSpriteFrames
mov  r8, r0
ldr  r5, =gGameStuff
adds r4, r3, #0
ldr  r0, =0x030060A0
lsls r1, r7, #5
adds r1, r1, r0
ldrb r2, [r5, #10]
```

The built code keeps the prologue `REG_DMA3` base pseudo live in a way that
copies it to `r4` too early, leaving only `r0/r1/r2` free in the width/height
block. That shifts the layer slot/mode scratch registers and cascades through
the rest of the function. Responsible pass remains the old gcc allocator
path (`local-alloc.c` / `global.c`), specifically live-range/equivalence
handling around the prologue DMA base pseudo; `loop.c` is not the main issue.

New levers ruled out in this round:
- Clean readable C with no pins: `byte_diff 590`, `diff_count 205`.
- Pin-free source rewrite with separate `cnt` and explicit slot local:
  `byte_diff 589`, `diff_count 203`.
- Required high-reg/base pins: improves to `byte_diff 449`, but lands on the
  same `r3`/`r4` live-range plateau.
- Newer `AGBCC_BIN` for this TU: worse, `byte_diff 515`, `diff_count 172`,
  with an extra stack spill around the loop index/count.
- Guarded `do` loop preheader shape: worse, `byte_diff 494`; it places
  frame/game loads nearer the target but spills `r9` through `[sp,#4]`.
- Unconditional frame/game hoist before the `for`: worse, `byte_diff 478`;
  loads move before the zero-count branch instead of after it.
- `register struct CharacterLayerCtrl *slot asm("r1")`: worse overall,
  `byte_diff 465`; it helps the local slot register but perturbs later
  case-body allocation.
- Current-tree corpus search found only sparse high-register examples
  (`metroidret/mf`, `pret/pokeruby`, `fireemblem8u`, `pokepinballrs`) and no
  source pattern that addresses this DMA-base preheader ordering.
- `corpus_asm_search.py` could not run the history pairing step because this
  worktree has no `tools/agent/corpus-mirrors` directory.

#include "game.h"
#include "gba/dma.h"
#include "gba/intr.h"
#include "gba/io.h"
#include "iwram.h"
#include "types.h"

#define CHARACTER_LAYER_BG0CNT 0x1C03
#define CHARACTER_LAYER_BG1CNT 0x1D02
#define CHARACTER_LAYER_BG2CNT 0x1E0D

struct CharacterSpriteFrame {
    const void *tilesA;
    const void *tilesB;
    const void *tilesC;
    u16 width;
    u16 height;
};

struct CharacterLayerCtrl {
    u8 _pad00[0x18];
    u16 width;
    u16 height;
    u8 _pad1C[4];
};

extern const struct CharacterSpriteFrame sCharacterSpriteFrames[];

extern void sub_0802D568(const void *src, void *dst);
extern void sub_0800EE34(u8 layer);
extern void sub_0800EE94(u8 layer);

void sub_0800E85C(u8 count)
{
    u16 zero;
    u8 pose;
    register GameStuff *game asm("r5");
    register const struct CharacterSpriteFrame *frames asm("r8");
    const struct CharacterSpriteFrame *frame;
    struct CharacterLayerCtrl *layerCtrl;
    register const void *tilesA asm("sl");
    register u8 cnt asm("r9");

    cnt = count;
    zero = 0;
    REG_DMA3.src = &zero;
    REG_DMA3.dst = (void *)0x02000000;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 2;
    (void)REG_DMA3.cnt;

    zero = 0;
    REG_DMA3.src = &zero;
    REG_DMA3.dst = (void *)0x06000000;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 2;
    (void)REG_DMA3.cnt;

    zero = 0;
    REG_DMA3.src = &zero;
    REG_DMA3.dst = (void *)0x05000000;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 2;
    (void)REG_DMA3.cnt;

    for (pose = 0; pose < cnt; pose++) {
        frames = sCharacterSpriteFrames;
        game = &gGameStuff;
        layerCtrl = (struct CharacterLayerCtrl *)0x030060A0;
        frame = &frames[game->pendingMode * 3 + pose];
        layerCtrl[pose].width = frame->width;

        frame = &frames[game->pendingMode * 3 + pose];
        layerCtrl[pose].height = frame->height;

        switch (pose) {
        case 0:
            frame = &frames[game->pendingMode * 3];
            if (frame->tilesA != NULL) {
                tilesA = frame->tilesA;
            }

            sub_0802D568(tilesA, (void *)0x02000000);
            REG_DMA3.src = (const void *)0x02000000;
            REG_DMA3.dst = (void *)0x06000000;
            REG_DMA3.cnt = DMA_ENABLE | 0x4000;
            (void)REG_DMA3.cnt;

            if (game->pendingMode == 13) {
                frame = &frames[game->pendingMode * 3];
                REG_DMA3.src = frame->tilesA;
                REG_DMA3.dst = (void *)0x06000000;
                REG_DMA3.cnt = DMA_ENABLE | 0x4000;
                (void)REG_DMA3.cnt;
            }

            frame = &frames[game->pendingMode * 3];
            sub_0802D568(frame->tilesC, (void *)0x02000000);
            REG_BG0CNT = CHARACTER_LAYER_BG0CNT;
            break;

        case 1:
            frame = &frames[game->pendingMode * 3 + 1];
            if (frame->tilesA != NULL) {
                REG_DMA3.src = frame->tilesA;
                REG_DMA3.dst = (void *)0x02010000;
                REG_DMA3.cnt = DMA_ENABLE | 0x2000;
                (void)REG_DMA3.cnt;
            }

            frame = &frames[game->pendingMode * 3 + 1];
            sub_0802D568(frame->tilesC, (void *)0x02010000);
            REG_BG1CNT = CHARACTER_LAYER_BG1CNT;
            break;

        case 2:
            frame = &frames[game->pendingMode * 3 + 2];
            if (frame->tilesA != NULL) {
                REG_DMA3.src = frame->tilesA;
                REG_DMA3.dst = (void *)0x0600C000;
                REG_DMA3.cnt = DMA_ENABLE | 0x1000;
                (void)REG_DMA3.cnt;
            }

            frame = &frames[game->pendingMode * 3 + 2];
            if (frame->tilesC != NULL) {
                REG_DMA3.src = frame->tilesC;
                REG_DMA3.dst = (void *)0x02020000;
                REG_DMA3.cnt = DMA_ENABLE | 0x8000;
                (void)REG_DMA3.cnt;
            } else {
                zero = 0;
                REG_DMA3.src = &zero;
                REG_DMA3.dst = (void *)0x02020000;
                REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x8000;
                (void)REG_DMA3.cnt;
            }

            REG_BG2CNT = CHARACTER_LAYER_BG2CNT;
            break;
        }
    }

    frames = sCharacterSpriteFrames;
    game = &gGameStuff;
    REG_DMA3.src = frames[game->pendingMode * 3].tilesB;
    REG_DMA3.dst = (void *)0x05000000;
    REG_DMA3.cnt = DMA_ENABLE | 0x100;
    (void)REG_DMA3.cnt;

    REG_DISPCNT = DISPCNT_OBJ_ON | DISPCNT_OBJ_1D;

    switch (cnt) {
    case 1:
        sub_0800EE94(0);
        sub_0800EE34(1);
        break;

    case 2:
        sub_0800EE94(0);
        sub_0800EE94(1);
        sub_0800EE34(2);
        sub_0800EE94(3);
        break;

    case 3:
        sub_0800EE94(0);
        sub_0800EE94(1);
        sub_0800EE94(2);
        sub_0800EE94(3);
        break;
    }

    REG_IE &= ~IRQ_HBLANK;
    REG_DISPSTAT &= ~DISPSTAT_HBLANK_IRQ;
    gIwram_3608._pad4 = 1;
    gIwram_3608._field_5 = 0;
}
