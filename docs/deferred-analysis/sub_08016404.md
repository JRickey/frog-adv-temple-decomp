## Round-23 progress (improves on round-22's 431 plateau)

Best clean result (NO pins/asm/CFLAGS): **byte_diff 375, diff_count 178,
size 572** (target 588). Down from round-22's 431/182/572.

The single new structural WIN vs round 22: **materialize TWO separate zeros**
for the prologue clears. The baserom does:
    movs r1, #0        ; zero A
    movs r5, #0        ; zero B
    strh r5, [clear,#2]
    strh r5, [clear,#0]
    strb r1, [active,#5]   ; uses zero A (r1), NOT r5
i.e. `active[5]=0` uses a DIFFERENT zero register than the two `clear[]` stores.
The fix (matches sibling sub_08016360's `int zero=0` idiom): put the clears and
`active[5]=z` in one block with a local `u8 z=0`:
    {
        u16 *clear = (u16 *)0x030065D0;
        u8 z = 0;
        clear[1] = 0;
        clear[0] = 0;
        active = (u8 *)0x03003608;
        active[5] = z;
    }
This alone took byte_diff 431 -> 375 (diff_count 182 -> 178). KEEP THIS.

## ROOT CAUSE — confirmed from the agbcc global-allocator dump

Reproduce: compile the .i with `-dg` and read `<i>.greg`:
    tools/preproc/preproc src/engine/sub_08016404.c charmap.txt \
      | cpp-15 -P -DREGION_US -nostdinc -Iinclude/ > /tmp/s.i
    tools/agbcc/bin/old_agbcc -o /tmp/s.s -Werror -O2 -mthumb-interwork \
      -fhex-asm -dg /tmp/s.i      # writes /tmp/s.i.greg

The constant `31` (used only in the two `fixed[i].hi = 31` -> `strh r3` stores)
is pseudo "Register 148": **refs = 5, live_length = 102, pref LO_REGS**. Its
QTY_CMP_PRI priority (~floor_log2(5)*5/102) is TINY because the live range spans
all of loop1 and there are only 5 refs. local-alloc.c additionally DOUBLES a
constant pseudo's live length (update_equiv_regs: `REG_LIVE_LENGTH *= 2`),
sinking the priority further. So `31` is allocated LAST among the loop-invariants
and, with no LOW reg free and the default REG_ALLOC_ORDER (r0..r7,r8,r9,sl,ip),
falls to **ip** -> the dump shows `insn 958: r3=31; insn 922: ip=r3`, then every
store reload-copies `mov rX,ip; strh rX`.

The baserom instead keeps `31` in a LOW reg (r3) directly and SPILLS `i<<3` to a
4th stack slot (`sub sp,#16`, `str r0,[sp,#12]`, reload `ldr r1,[sp,#12]; add
r1,ip`) so `&bounds[i]` is recomputed per half-iteration. The whole 375-byte
diff cascades from this one choice:
    baserom:  lo->r8,  hi<<16->r9,  slo->sl, shi->r9(reuse), bounds->ip, 31->r3,  fixed->r7
    mine:     lo->r7,  bounds->r8,  slo->sl, shi->r9,        31->ip,     fixed->r4
Every diff is a register-NAME mismatch; the opcode+immediate SEQUENCE is correct
(size 572 vs 588 is purely the missing spill machinery + the extra ip<->low
copies cancelling out).

The shared computation the baserom keeps alive from the PROLOGUE through the
early-return check AND the DMA block INTO loop1: `(viewX-0x30)<<16` (r4) and
`hi<<16` (r9). loop1 derives slo = r4>>19 and shi = r9>>19 at the preheader.
Any source change that makes slo/shi materialize earlier (or that frees a low
reg) breaks this and regresses.

#include "macros.h"
#include "types.h"
#include "game.h"
#include "gba/dma.h"

/* One screen-edge clip span: a tile range [lo, hi] derived from a pixel range.
 * Two fixed entries at 0x030063C0 hold the vertical clip; the `count` entries
 * at 0x03006360 hold the per-column horizontal clip. Same record as sibling
 * sub_08016360's Dst08016360 (base 0x03006360). */
typedef struct {
    u16 lo;
    u16 hi;
    u8 unk4;
    u8 unk5;
    u8 _unk6[2];
    u32 unk8;
    u8 unk12;
    u8 _unk13[3];
} ClipSpan;

/* Source record (ROM): a pixel range [start, end] per column. */
typedef struct {
    u16 start;
    u16 end;
    u16 _unk4;
    u16 _unk6;
} ColumnRange;

/* Bounds for the two fixed vertical spans, in pixels. */
typedef struct {
    u16 _unk0;
    u16 top;
    u16 _unk4;
    u16 _unk6;
} ViewBounds;

/* Camera/scroll state; +8 is the current view X (in pixels). */
typedef struct {
    u8 _pad0[8];
    s32 viewX;
} ViewState;

#define gViewState ((ViewState *)0x030060A0)
#define gViewBounds ((ViewBounds *)0x0830696C)
#define gFixedSpans ((ClipSpan *)0x030063C0)
#define gColumnSpans ((ClipSpan *)0x03006360)

void sub_08016404(ColumnRange *columns, u8 count)
{
    vu16 fill;
    u8 *active;
    s32 viewX;
    ViewState *vs;
    u16 lo, hi;
    u16 slo, shi;
    ViewBounds *bounds;
    ClipSpan *fixed;
    u8 i;

    {
        u16 *clear = (u16 *)0x030065D0;
        u8 z = 0;
        clear[1] = 0;
        clear[0] = 0;
        active = (u8 *)0x03003608;
        active[5] = z;
    }

    viewX = gViewState->viewX - 0x30;
    lo = viewX;
    hi = lo + 0x100;

    {
        ViewBounds *vb = gViewBounds;
        if (lo >= vb[0].top)
            goto reset_columns;
        if (hi <= vb[1]._unk0)
            goto reset_columns;
    }

    fill = 0;
    REG_DMA3.src = (void *)&fill;
    REG_DMA3.dst = (void *)0x030053A0;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x50;
    (void)REG_DMA3.cnt;

    fill = 0;
    REG_DMA3.src = (void *)&fill;
    REG_DMA3.dst = (void *)0x03003610;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x50;
    (void)REG_DMA3.cnt;

    active[5] = 1;
    active[4] = 1;

    slo = lo >> 3;
    shi = hi >> 3;

    bounds = gViewBounds;
    fixed = gFixedSpans;

    for (i = 0; i <= 1; i++) {
        if (slo < (u16)(bounds[i]._unk0 >> 3))
            fixed[i].lo = (((s32)(u16)bounds[i]._unk0 - lo) >> 3) + 1;
        else
            fixed[i].lo = 0;

        if (shi < (u16)(bounds[i].top >> 3)) {
            fixed[i].hi = 31;
        } else {
            fixed[i].hi = ((s32)(u16)bounds[i].top - lo) >> 3;
            if ((s16)fixed[i].hi > 31)
                fixed[i].hi = 31;
        }
    }

    vs = gViewState;
    for (i = 0; i < count; i++) {
        u8 *base;
        s32 x;
        u16 j;

        viewX = vs->viewX;
        if (viewX >= (u16)columns[i].end)
            continue;
        if (viewX + 0x100 <= (u16)columns[i].start)
            continue;

        if (viewX > (u16)columns[i].start)
            gColumnSpans[i].lo = 0;
        else
            gColumnSpans[i].lo = columns[i].start - viewX;

        viewX = vs->viewX;
        if (viewX + 0x100 < (u16)columns[i].end)
            gColumnSpans[i].hi = 255;
        else
            gColumnSpans[i].hi = columns[i].end - viewX;

        x = gGameStuff._unk00 - gColumnSpans[i].unk8;
        if (x >= gColumnSpans[i].unk12) {
            gColumnSpans[i].unk8 = gGameStuff._unk00;
            if (gColumnSpans[i].unk5 == 0)
                gColumnSpans[i].unk4++;
            else
                gColumnSpans[i].unk4--;
        }

        for (j = gColumnSpans[i].lo; j < (s16)gColumnSpans[i].hi; j++) {
            base = (gColumnSpans[i].unk5 & 0x10) ? (u8 *)0x03003610 : (u8 *)0x030053A0;
            base[j] = gColumnSpans[i].unk4;
        }
    }

reset_columns:
    fill = 0;
    REG_DMA3.src = (void *)&fill;
    REG_DMA3.dst = (void *)0x03005440;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x30;
    (void)REG_DMA3.cnt;

    fill = 0;
    REG_DMA3.src = (void *)&fill;
    REG_DMA3.dst = (void *)0x030036B0;
    REG_DMA3.cnt = DMA_ENABLE | DMA_SRC_FIXED | 0x30;
    (void)REG_DMA3.cnt;
}
