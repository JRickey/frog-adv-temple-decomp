## What it does (logic fully understood — verified against the disasm)

Per-frame attract-mode script-step advancer. Called from sub_080004C4
(init1.c) on the mode-24 attract path; its (u16-cast) return is the live
opcode. Reads from a per-mode u16 opcode stream selected by
gGameStuff.pendingMode-1 from the ROM pointer table at 0x08307fec
(`gScriptPtrTable_08307fec`, added to linker.ld). Uses gIwram_34C0 as its
state: cursor (+0), delay (+1), stepTick (+4), holdFlag (+12). Opcodes:
0xFE sets holdFlag, 0xFD clears it, 0xFF resets gGameStuff.mode=4 +
gIwram_3480._data[0]=4, 0 skips to the next entry (re-arms delay/stepTick),
anything else is returned as the live opcode. The delay/stepTick form a
60-tick (`__udivsi3(now - stepTick, 60)`) gate via sub_08000900().

Struct header IwramAt34C0 was grown (cursor/delay/stepTick/lastTick/holdFlag)
— that part is correct and should be KEPT on the next attempt.

## Best-effort C (stable, byte_diff 152 / diff_count 39 / size 204)

```c
#include "game.h"
#include "iwram.h"
#include "macros.h"
#include "types.h"

/* Per-mode attract/cutscene script table at 0x08307fec; each entry points to
 * a u16 opcode stream advanced by the cursor in gIwram_34C0. */
extern const u16 *const gScriptPtrTable_08307fec[];

extern u32 sub_08000900(void);
extern u32 __udivsi3(u32 num, u32 den);

int sub_080179B8(void)
{
    struct IwramAt34C0 *s;
    register const u16 *script asm("r5");
    register int op asm("r4");
    int cursor;

    s = &gIwram_34C0;
    if (s->delay != 0) {
        if (__udivsi3(sub_08000900() - s->stepTick, 60) < s->delay) {
            return 0;
        }
    }

    s->delay = 0;
    s->stepTick = 0;

    script = gScriptPtrTable_08307fec[gGameStuff.pendingMode - 1];
    op = script[s->cursor];
    if (op == 0xFE) {
        s->holdFlag = 1;
    } else if (op == 0xFD) {
        s->holdFlag = 0;
    }

    if (gIwram_35E0._field_10 & 1) {
        if (gIwram_34C0.holdFlag != 0) {
            return 0;
        }
    }

    cursor = s->cursor;
    s->cursor = cursor + 1;
    op = script[(u8)cursor];
    if (op == 0) {
        s->cursor = cursor + 2;
        s->delay = (u8)script[(u8)(cursor + 1)];
        s->stepTick = sub_08000900();
        return op;
    }

    if ((u16)(op - 0xFD) <= 1) {
        return 0;
    }
    if (op != 0xFF) {
        return op;
    }

    gGameStuff.mode = 4;
    gIwram_3480._data[0] = 4;
    return 0;
}
```
