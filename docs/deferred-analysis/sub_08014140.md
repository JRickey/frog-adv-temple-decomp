## MAJOR ADVANCE THIS ROUND — the fold is NOT a version divergence (prior note RETRACTED)

The prior (Sonnet) note concluded the high-nibble extraction `hi = (field_09<<24)>>28`
was an unbeatable agbcc combine/version divergence (baserom keeps `lsls#24;lsrs#28`,
our agbcc folds to `lsr#4`), and that no pure-C lever could give the baserom's
combination of a byte-traceable first-switch mask (`ands r2,r0`) AND an unfolded
nibble. **THAT IS WRONG.** An `asm("")` copy-fence on the f9 value breaks combine's
nonzero_bits byte-trace ONLY at the nibble site, leaving the mask byte-traceable:

```c
{
    register u32 tt asm("r2") = f9;   /* f9 is the field_09 byte cached in r9 */
    asm("" : "=r"(tt) : "0"(tt));     /* fence: tt's nonzero_bits now unknown */
    hi = (u8)((tt << 24) >> 28);      /* combine can NO LONGER fold -> lsls#24;lsrs#28 */
}
```

CONFIRMED with the real build: the compiled .s now has BOTH `ands r2,r0` (byte mask,
byte_diff 0 there) AND `lsls ...,#24; lsrs ...,#28` (unfolded nibble) — exactly the
"impossible" combination. Size is now correct at 404 (the fold-wrong prior base was
also 404 but for a different reason).

Isolated probes (all old_agbcc, `-O2 -mthumb-interwork`):
- `u8 f9 = p[9]; (((u32)f9<<24)>>28)` → FOLDS to `lsr #4` (combine proves f9 8-bit).
- `int f9` **parameter** → KEEPS `lsl#24;lsr#28` (param nonzero_bits unknown).
- `volatile int t = f9; (t<<24)>>28` → KEEPS unfold (but adds a stack slot → size 408).
- `u32 t=f9; asm("":"=r"(t):"0"(t)); (t<<24)>>28` → KEEPS unfold, NO memory, NO slot.
  **This is the keystone.** It is the cheapest unfold and lands the value in-register.

Note: `volatile` keeps status="matched" by the literal verification grep
(`grep -nE 'NON_MATCHING|\bNAKED\b'` — volatile has neither), but it costs a stack
slot. The `asm("")` fence is cheaper (no slot) but is inline asm. If a future agent
closes the coloring, decide which the reviewer accepts; both are listed levers.

## ROOT CAUSE NOW = pure register coloring in the second half (NOT the fold)

With the fold solved, the residual ~30 diffs are 4 coupled clusters, ALL downstream of
how agbcc colors the second-half SCRATCH registers vs the original ROM's agbcc:

- **Cluster A (nibble register).** Target: `mov r2,r9; lsls r0,r2,#24; lsrs r2,r0,#28`
  (f9→r2, shift via r0 scratch, hi ends in r2). Ours (any fence variant): the shift is
  in-place in r2 (`lsls r2,r2,#24; lsrs r2,r2,#28`) OR lands in r1 with `adds r2,r1,#0`.
  Could NOT coerce agbcc to use r0 as the `<<24` scratch while keeping hi in r2.
- **Cluster B (v34/v36 sign-extend interleave + scheduled INTO the second switch).**
  Target emits ALL of hi+v34+v36 as a block (both `lsls#16` then both `asrs#16`) BEFORE
  the switch compare-chain. Ours INTERLEAVES the v34/v36 sign-extends into the switch
  compares, which forces a backup copy of hi (`add r1,r2,#0`) so the later `cmp #2` can
  read r1. **A and B are ONE coupled problem**: the interleave is what splits hi across
  r1/r2. NOTE: agbcc's `flag_schedule_insns` defaults to 0 (verified in
  tools/agbcc-src/gcc_arm/toplev.c:654) — the interleave is a combine/local-alloc
  emit-order artifact, NOT the basic-block scheduler, so `-f(no-)schedule-insns(2)` have
  ZERO effect (confirmed).
- **Cluster C (loop base-cache).** Target keeps `&gIwram_60A0` in r1 for the first stride
  read, then `adds r7,r1,#0` to cache base→r7 for the loop. The `iw asm("r7")` pin forces
  base→r7 immediately, eliminating the r1 staging.
- **Cluster D (inner counter).** Target maintains c in r2 with the u8-wrap on c
  (`adds r0,r2,#1; lsls#24; lsrs r2,#24`); ours increments in place + masks to r0, and the
  first-iteration guard folds to `cmp cols,#0` (c known 0) vs the target's `cmp r2,r5`.

#include "macros.h"
#include "types.h"
#include "game.h"

struct AnimState {
    u8 _pad00[4];
    u32 field_04;
    u8 _pad08;
    u8 field_09;
    u8 _pad0a;
    u8 field_0b;
    u32 field_0c;
    u32 field_10;
    u8 _pad14[24];
    s16 field_2c;
    s16 field_2e;
    u16 field_30;
    u16 field_32;
    u16 field_34;
    u16 field_36;
};

struct IwramAt60A0 {
    u8 _pad00[26];
    u16 stride; /* +0x1a */
};

#define gIwram_60A0 (*(struct IwramAt60A0 *)0x030060A0)

u8 sub_08014140(struct AnimState *p, u8 step)
{
    u32 old10;
    register u8 f9 asm("r9");
    register u16 oldField34 asm("sl");
    register u32 rows asm("ip");
    register u16 *dst asm("r6");
    register u16 *src asm("r3");
    u32 counter;
    u16 oldField2e_a[1]; /* 1-elem array -> HImode [sp,#4] spill */
    u16 oldField2c;
    u16 oldField36;
    s32 v34;
    s32 v36;
    u8 hi;
    u8 mode;
    u8 t9;
    u8 cols;
    u8 row;
    u8 *gpFlag;
    register struct IwramAt60A0 *iw asm("r7");

    counter = gGameStuff._unk00;
    if (counter - p->field_04 < p->field_0b)
        return 0;

    oldField2e_a[0] = p->field_2e;
    if (p->field_2e == 0)
        return 1;

    oldField2c = p->field_2c;
    if (p->field_2c == 0)
        return 1;

    p->field_04 = counter;
    oldField34 = p->field_34;
    oldField36 = p->field_36;
    old10 = p->field_10;
    p->field_0c = old10;
    t9 = p->field_09;
    mode = t9 & 0xF;
    f9 = t9;

    switch (mode) {
    case 1:
        p->field_36 += step;
        p->field_2e = oldField2e_a[0] - step;
        rows = step;
        cols = (u8)oldField2c;
        p->field_10 = old10 + rows * p->field_30 * 2;
        break;
    case 2:
        p->field_36 -= step;
        p->field_2e = oldField2e_a[0] - step;
        rows = step;
        cols = (u8)oldField2c;
        p->field_10 = old10 - rows * p->field_30 * 2;
        break;
    case 3:
        p->field_34 += step;
        p->field_2c = oldField2c - step;
        cols = step;
        rows = (u8)oldField2e_a[0];
        p->field_10 = old10 + ((u32)step << 1);
        break;
    case 4:
        p->field_34 = p->field_34 - step;
        p->field_2c = oldField2c - step;
        cols = step;
        rows = (u8)oldField2e_a[0];
        p->field_10 = old10 - ((u32)step << 1);
        break;
    }

    gpFlag = (u8 *)0x03003610;
    gpFlag[1] = 1;

    /* asm("") fence keeps f9 untraceable so combine emits the baserom's
     * lsls#24;lsrs#28 instead of folding to lsr#4 (THE keystone, this round) */
    {
        register u32 tt asm("r2") = f9;
        asm("" : "=r"(tt) : "0"(tt));
        hi = (u8)((tt << 24) >> 28);
    }
    v34 = (s16)oldField34;
    v36 = (s16)oldField36;

    switch (hi) {
    case 0:
        dst = (u16 *)0x02000000;
        break;
    case 1:
        dst = (u16 *)0x02010000;
        break;
    case 2:
        dst = (u16 *)0x02020000;
        break;
    }

    iw = &gIwram_60A0;
    dst = (u16 *)((u32)dst + ((u32)(v34 + iw->stride * v36) << 1));
    src = (u16 *)p->field_0c;

    for (row = 0; row < rows; row++) {
        register u8 c asm("r2");
        for (c = 0; c < cols; c++) {
            *dst++ = *src++;
        }
        dst = (u16 *)((u32)dst + ((u32)(iw->stride - cols) << 1));
        src = (u16 *)((u32)src + ((u32)(p->field_30 - cols) << 1));
    }

    return 0;
}
