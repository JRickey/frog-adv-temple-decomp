# sub_08014140 — deferred analysis

2D tilemap-blit / scroll-step on the `struct AnimState` shared with
`sub_080140A0` (sub_08014078.c). Pure-C structure is **fully solved** — every
instruction is the correct opcode; the only residual is one agbcc spill-mode
artifact plus the register-coloring cascade it causes.

## What it does
`u8 sub_08014140(struct AnimState *p, u8 step)`:
1. Frame gate: `if (gGameStuff._unk00 - p->field_04 < p->field_0b) return 0;`
2. Save `oldField2e = p->field_2e`; `if (field_2e == 0 || field_2c == 0) return 1;`
   (each compared via signed `ldrsh`, the save via unsigned `ldrh`).
3. Commit: `field_04 = counter`, snapshot `field_34/36/10`, `field_0c = field_10`.
4. `switch (field_09 & 0xF)` (scroll mode 1..4): updates field_34/36/2c/2e and
   recomputes `field_10` (dest offset) by ± `step*field_30*2` or ± `step*2`.
   Also sets `cols`/`rows` (the blit extents) and `*(u8*)0x03003611 = 1`.
5. `switch ((field_09 >> 4) & 0xF)` (dest bank 0/1/2) selects EWRAM base
   0x02000000 / 0x02010000 / 0x02020000.
6. `dst = base + (field_34_signed + gIwram_60A0.stride*field_36_signed)*2`,
   `src = field_0c`. Then a `rows × cols` u16 2D copy with src row-stride
   `field_30` and dest row-stride `gIwram_60A0.stride` (gIwram_60A0 + 0x1a).
7. `return 0;`

## Drift (best pure-C: byte_diff 260, diff_count 95; size 404 = correct)
The function is structurally exact. The blocker is a single agbcc spill-mode
decision that cascades into register coloring:

- **`oldField2e` is spilled as SImode (word) instead of HImode (halfword).**
  Target: `ldrh r0,[r4,#46]; mov r7,sp; strh r0,[r7,#4]` at the save, and
  `mov r5,sp; ldrh r5,[r5,#4]` reloaded in EACH switch case (a genuine HImode
  frame variable reloaded per use). agbcc instead keeps an SImode pseudo,
  CSE-caches the widened value, and reload-spills it as a WORD: `str r7,[sp,#4]`
  / `ldr r7,[sp,#4]`. `oldField36` IS correctly SImode/word in both (`str/ldr
  [sp,#0]`), so the two slots differ ONLY in mode for oldField2e.
- Once the spill mode is fixed (proven via an `asm volatile("" : "+m"(x))`
  barrier — see below), **every remaining diff is pure register coloring**
  (r1↔r2↔r3↔r7 temp choices, and the slot-offset [sp,#0]↔[sp,#4] swap between
  oldField2e/oldField36). With the barrier the readable C reaches **byte_diff
  183, ALL `ARGUMENT_MISMATCH`** (no insertion/deletion/replacement).

#include "macros.h"
#include "types.h"
#include "game.h"

struct AnimState {
    u8 _pad00[4];
    u32 field_04;
    u8 _pad08;
    u8 field_09;
    u8 _pad0a;
    u8 field_0b;
    u32 field_0c;
    u32 field_10;
    u8 _pad14[24];
    s16 field_2c;
    s16 field_2e;
    u16 field_30;
    u16 field_32;
    u16 field_34;
    u16 field_36;
};

struct IwramAt60A0 {
    u8 _pad00[26];
    u16 stride; /* +0x1a */
};

extern struct IwramAt60A0 gIwram_60A0;

u8 sub_08014140(struct AnimState *p, u8 step)
{
    register u32 old10 asm("r8");
    register u8 f9 asm("r9");
    register u16 oldField34 asm("sl");
    register u32 rows asm("ip");
    register u16 *dst asm("r6");
    register u16 *src asm("r3");
    register u32 counter asm("r2");
    u16 oldField2e;
    u16 oldField2c;
    u16 oldField36;
    s16 v34;
    s16 v36;
    u8 hi;
    u8 mode;
    u8 t9;
    u8 cols;
    u8 row;

    counter = gGameStuff._unk00;
    if (counter - p->field_04 < p->field_0b)
        return 0;

    oldField2e = p->field_2e;
    if (p->field_2e == 0)
        return 1;

    oldField2c = p->field_2c;
    if (p->field_2c == 0)
        return 1;

    p->field_04 = counter;
    oldField34 = p->field_34;
    oldField36 = p->field_36;
    old10 = p->field_10;
    p->field_0c = old10;
    t9 = p->field_09;
    mode = t9 & 0xF;
    f9 = t9;

    switch (mode) {
    case 1:
        p->field_36 += step;
        p->field_2e = oldField2e - step;
        rows = step;
        cols = (u8)oldField2c;
        p->field_10 = old10 + (u32)((u16)(step * p->field_30) << 1);
        break;
    case 2:
        p->field_36 -= step;
        p->field_2e = oldField2e - step;
        rows = step;
        cols = (u8)oldField2c;
        p->field_10 = old10 - (u32)((u16)(step * p->field_30) << 1);
        break;
    case 3:
        p->field_34 += step;
        p->field_2c = oldField2c - step;
        cols = step;
        rows = (u8)oldField2e;
        p->field_10 = old10 + ((u32)step << 1);
        break;
    case 4:
        p->field_34 = oldField34 - step;
        p->field_2c = oldField2c - step;
        cols = step;
        rows = (u8)oldField2e;
        p->field_10 = old10 - ((u32)step << 1);
        break;
    }

    *(u8 *)0x03003611 = 1;

    hi = (f9 << 24) >> 28;
    v34 = (s16)oldField34;
    v36 = (s16)oldField36;

    switch (hi) {
    case 0:
        dst = (u16 *)0x02000000;
        break;
    case 1:
        dst = (u16 *)0x02010000;
        break;
    case 2:
        dst = (u16 *)0x02020000;
        break;
    }

    dst = (u16 *)((u32)dst + ((u32)(v34 + gIwram_60A0.stride * v36) << 1));
    src = (u16 *)p->field_0c;

    for (row = 0; row < rows; row++) {
        u8 c;
        for (c = 0; c < cols; c++) {
            *dst++ = *src++;
        }
        dst = (u16 *)((u32)dst + ((u32)(gIwram_60A0.stride - cols) << 1));
        src = (u16 *)((u32)src + ((u32)(p->field_30 - cols) << 1));
    }

    return 0;
}
