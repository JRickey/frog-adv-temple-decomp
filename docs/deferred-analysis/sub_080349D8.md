## Drift (the residual 35, three coupled coalescing clusters)

agbcc inserts a register *copy* where the baserom uses the value in place —
the classic permuter-territory coalescing residual. permuter is the right
tool but its venv is MISSING repo-wide (vendor/decomp-permuter symlink → an
empty submodule mountpoint in main; bootstrap printed "WARN: permuter .venv
missing"). No -fXXX flag moved any of these (swept: -fno-gcse[worse],
-fno-cse-follow-jumps, -fno-cse-skip-blocks, -fno-schedule-insns/2,
-fno-expensive-optimizations, -fforce-mem, -fno-peephole, -fno-thread-jumps).

1. **Rounding mask (`ands` operand order).** baserom `ands r1, r0` keeps the
   0xFF mask live in r0 so 0x100 = `adds r0,#1`; built `ands r0, r1`
   destroys the mask so 0x100 = `movs 0x80; lsls #1`. This is a commutative-&
   reload decision. `register u32 guard asm("r1")` DOES flip it to `ands
   r1,r0` + the `adds r0,#1` 0x100 reuse + the inner `orrs` — but then agbcc
   CSEs `(u32)frac` into r3 (`adds r3,r4,#0; adds r1,r3,#0`) where baserom
   re-copies r4→r1 twice. The two are coupled: no-pin → no r3 copy but wrong
   `ands`; pin → right `ands`+0x100 but +1 r3 copy. Both net diff_count 35;
   no-pin nets lower byte_diff (94 vs 106). NEXT: permuter from the guard-pin
   (106) base should drop the r3 copy and reach 0 — the rounding is otherwise
   identical to baserom.
2. **Overflow high-word compare.** baserom `cmp r5, r0` (frac_high in r5
   direct); built copies `adds r0,r5,#0; cmp r0,r2`. `(u32)(frac>>32)`
   extraction lands in a fresh temp. (Also a stray `movs r0,r0` pool-align
   nop appears.)
3. **Assemble word stores.** baserom writes frac_low/frac_high SP-relative
   (`str r4,[sp+0]`, `ands r2,r5`) then materializes `mov r2, sp` LATE for
   the strh/strb; built holds the pinned `p`(r2) and uses `[r2,#N]` for all,
   and copies r5→r0 for frac_high. Moving `p=&result` after the word writes
   re-promotes the union (153). The word writes want SP-relative while the
   narrow writes want the held pointer — agbcc can't split it the way GCC's
   union return does.

## Best-effort C (byte_diff 94; the guard-pin variant reaches the correct
## rounding shape at 106 — see Drift #1)

Requires `src/game/sub_080349d8.s: CFLAGS += -mno-thumb-interwork` in the
Makefile (added in the fp-bit cluster block) and the linker.ld asm-slice
removal.

```c
#include "macros.h"
#include "types.h"

struct DoubleComponents {
    u32 type;
    u32 sign;
    s32 exp;
    u64 frac64;
};

union DoubleBits {
    u64 value;
    u32 words[2];
    struct {
        u32 frac_low;
        u32 frac_high : 20;
        u32 exp : 11;
        u32 sign : 1;
    } bits;
};

u64 sub_080349D8(struct DoubleComponents *comp)
{
    union DoubleBits result;
    register union DoubleBits *p asm("r2");
    u64 frac;
    u32 biased_exp;
    s32 exp;
    s32 shift;
    u32 sign;
    u32 c0;
    u32 c1;
    u32 c2;
    register u32 sticky asm("r2");
    u32 guard;
    u32 lo;
    u32 hi;

    frac = comp->frac64;
    sign = comp->sign;
    biased_exp = 0;

    c0 = 0;
    if (comp->type <= 1)
        c0 = 1;
    if (c0) {
        biased_exp = 0x7FF;
        frac |= (u64)0x80000 << 32;
        goto assemble;
    }

    c1 = 0;
    if (comp->type == 4)
        c1 = 1;
    if (c1)
        goto infinity;

    c2 = 0;
    if (comp->type == 2)
        c2 = 1;
    if (c2) {
        frac = 0;
        goto assemble;
    }

    if (frac == 0)
        goto assemble;

    exp = comp->exp;
    if (exp < -1022) {
        shift = -1022 - exp;
        if (shift > 56) {
            frac = 0;
            goto shift_out;
        }
        frac = frac >> shift;
        goto shift_out;
    }

    if (exp > 1023) {
    infinity:
        biased_exp = 0x7FF;
        frac = 0;
        goto assemble;
    }

    biased_exp = exp + 0x3FF;

    guard = (u32)frac & 0xFF;            /* DRIFT#1: register u32 guard asm("r1")
                                            flips ands→r1,r0 + 0x100 reuse */
    sticky = 0;
    if (guard == 0x80 && sticky == 0) {
        if ((((u32)frac & 0x100) | sticky) != 0)
            frac += 0x80;
    } else {
        frac += 0x7F;
    }

    if ((u32)(frac >> 32) > 0x1FFFFFFF) {
        frac = frac >> 1;
        biased_exp += 1;
    }

shift_out:
    frac = frac >> 8;

assemble:
    p = &result;
    p->bits.frac_low = (u32)frac;
    p->bits.frac_high = (u32)(frac >> 32) & 0xFFFFF;
    p->bits.exp = biased_exp & 0x7FF;
    p->bits.sign = sign;

    /* FLOAT_WORD_ORDER swap; the memory clobber stops agbcc collapsing it. */
    lo = p->words[0];
    hi = p->words[1];
    p->words[0] = hi;
    p->words[1] = lo;
    asm("" : : : "memory");
    return p->value;
}
```
